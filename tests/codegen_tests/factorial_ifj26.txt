import ifj26.*

@main def run(): Unit =
    val n: Int = 6
    if n < 0 then
        print("Factorial is not defined for negative numbers\n")
    else
        var f: Int = 1
        var i: Int = 1
        while i <= n do
            f = f * i
            i = i + 1
        print(f)
        print("\n")
