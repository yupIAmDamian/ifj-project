import ifj26.*

@main def run(): Unit =
    val n: Int = readInt()
    val k: Int = 5 + calc(5)
    print(k)
    if n < 0 then
        print("Factorial is not defined for negative numbers\n")
    else
        var f: Int = 1
        var i: Int = 1
        while i <= n do
            f = f * i
            i = i + 1
        print(f)
        print("\n")
def calc(x: Int): Int =
    return x + 8 * 7
